cmake_minimum_required(VERSION 3.16)
project(swerve_chassis_controller)
find_package(ament_cmake REQUIRED)
find_package(swerve_chassis_core REQUIRED)
foreach(dep rclcpp sensor_msgs std_msgs geometry_msgs nav_msgs tf2_ros swerve_chassis_msgs)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(swerve_controller src/controller.cpp)
target_compile_features(swerve_controller PRIVATE cxx_std_17)
target_compile_options(swerve_controller PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(swerve_controller swerve_chassis_core::swerve_chassis_core)
ament_target_dependencies(swerve_controller rclcpp sensor_msgs std_msgs geometry_msgs nav_msgs tf2_ros swerve_chassis_msgs)
install(TARGETS swerve_controller DESTINATION lib/${PROJECT_NAME})
ament_package()
