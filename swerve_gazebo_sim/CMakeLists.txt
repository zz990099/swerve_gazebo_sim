cmake_minimum_required(VERSION 3.16)
project(swerve_gazebo_sim)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY config launch urdf worlds DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/ground_truth_localization.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  foreach(suite model launch truth controller localization)
    ament_add_pytest_test(${suite} test/test_${suite}.py TIMEOUT 90)
  endforeach()
endif()
ament_package()
